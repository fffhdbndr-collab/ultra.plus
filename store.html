import { supabase } from './supabase.js' let products = [] let cart = [] const cartElement = document.getElementById("cart"); const overlay = document.getElementById("overlay"); const cartItems = document.getElementById("cartItems"); const cartCount = document.getElementById("cartCount"); const cartTotal = document.getElementById("cartTotal"); const toast = document.getElementById("toast"); const searchBox = document.getElementById("searchBox"); const searchInput = document.getElementById("searchInput"); const loginWrap = document.getElementById("loginWrap"); let toastTimer; function showToast(message){ toast.textContent = message; toast.classList.add("show"); clearTimeout(toastTimer); toastTimer = setTimeout(() => { toast.classList.remove("show"); },2200); } // جلب المنتجات والباقات مباشرة من جدول products في سوبابيس async function fetchProducts() { const { data, error } = await supabase.from('products').select('*') if (error) { console.error('خطأ في جلب الباقات:', error) return } if (data && data.length > 0) { products = data renderProducts() } } // تحديث عرض المنتجات في الواجهة ديناميكياً function renderProducts() { const container = document.querySelector(".products") if (!container) return container.innerHTML = "" products.forEach((product) => { const isGold = product.name.includes("الذهبية") container.innerHTML += `
${isGold ? 'الأكثر تميزاً' : 'الباقة الفضية'}
ULTRA PLUS | ألترا بلس
${product.name}
${product.description || ''}
✓ مدة الاشتراك: ${product.duration_days || 0} يوم
${Number(product.price).toFixed(2)} ر.س
` }) // إعادة ربط أزرار الإضافة للسلة بعد تحديث الـ HTML document.querySelectorAll(".add-btn").forEach(button => { button.addEventListener("click", () => { addToCart(button.dataset.productId); }); }); } function addToCart(id){ const product = products.find(p => p.id == id); if(!product) return; cart.push(product); updateCart(); showToast("تمت إضافة الباقة للسلة"); openCart(); } function removeFromCart(index){ cart.splice(index,1); updateCart(); } function updateCart(){ cartCount.textContent = cart.length; if(cart.length === 0){ cartItems.innerHTML = `
السلة فارغة
`; cartTotal.textContent = "0.00"; return; } cartItems.innerHTML = ""; let total = 0; cart.forEach((item,index) => { total += Number(item.price); cartItems.innerHTML += `
${item.name.includes("الذهبية") ? "★" : "＋"}
${item.name} ${Number(item.price).toFixed(2)} ر.س
`; }); cartTotal.textContent = total.toFixed(2); document.querySelectorAll("[data-remove]").forEach(button => { button.addEventListener("click",() => { removeFromCart(Number(button.dataset.remove)); }); }); } function openCart(){ cartElement.classList.add("show"); overlay.classList.add("show"); } function closeCart(){ cartElement.classList.remove("show"); overlay.classList.remove("show"); } function checkout(){ if(cart.length === 0){ showToast("السلة فارغة"); return; } showToast("سيتم ربط بوابة الدفع الحقيقية لاحقاً"); } function toggleSearch(){ searchBox.classList.toggle("show"); if(searchBox.classList.contains("show")){ setTimeout(() => { searchInput.focus(); },100); } } function searchProducts(value){ const query = value.trim().toLowerCase(); document.querySelectorAll(".product").forEach((card) => { const name = card.querySelector(".product-name").textContent.toLowerCase(); const desc = card.querySelector(".product-desc").textContent.toLowerCase(); const found = name.includes(query) || desc.includes(query); card.style.display = found ? "" : "none"; }); } function scrollToSection(id){ const section = document.getElementById(id); if(!section) return; section.scrollIntoView({ behavior:"smooth", block:"start" }); } function setActive(button){ document.querySelectorAll(".nav-links button") .forEach(btn => btn.classList.remove("active")); if(button){ button.classList.add("active"); } } function toggleMobileMenu(){ const links = document.querySelector(".nav-links"); if(links.classList.contains("mobile-open")){ links.classList.remove("mobile-open"); links.style.display = ""; return; } links.classList.add("mobile-open"); links.style.position = "fixed"; links.style.top = "68px"; links.style.right = "4%"; links.style.left = "4%"; links.style.padding = "9px"; links.style.background = "#091113"; links.style.border = "1px solid rgba(32,215,193,.13)"; links.style.borderRadius = "14px"; links.style.display = "flex"; links.style.flexDirection = "column"; links.style.zIndex = "1100"; } function openLogin(){ loginWrap.classList.add("show"); } function closeLogin(){ loginWrap.classList.remove("show"); } document.querySelectorAll("[data-action]").forEach(button => { button.addEventListener("click",() => { const action = button.dataset.action; if(action === "home") scrollToSection("home"); if(action === "products") scrollToSection("products"); if(action === "features") scrollToSection("features"); if(action === "faq") scrollToSection("faq"); if(button.closest(".nav-links")) setActive(button); }); }); document.getElementById("accountBtn").addEventListener("click",openLogin); document.getElementById("closeLogin").addEventListener("click",closeLogin); loginWrap.addEventListener("click",event => { if(event.target === loginWrap) closeLogin(); }); document.getElementById("searchBtn").addEventListener("click",toggleSearch); searchInput.addEventListener("input",event => { searchProducts(event.target.value); }); document.getElementById("cartBtn").addEventListener("click",openCart); document.getElementById("closeCart").addEventListener("click",closeCart); overlay.addEventListener("click",closeCart); document.getElementById("checkoutBtn").addEventListener("click",checkout); document.getElementById("mobileBtn").addEventListener("click",toggleMobileMenu); document.querySelectorAll(".faq-question").forEach(button => { button.addEventListener("click",() => { button.parentElement.classList.toggle("open"); }); }); document.getElementById("loginSubmit").addEventListener("click",async () => { const email = document.getElementById("loginEmail").value.trim(); const password = document.getElementById("loginPassword").value.trim(); if(!email || !password){ showToast("اكتب البريد وكلمة المرور"); return; } const { error } = await supabase.auth.signInWithPassword({ email, password }) if (error) { showToast('خطأ في تسجيل الدخول: ' + error.message) return } showToast('تم تسجيل الدخول بنجاح!') setTimeout(() => location.reload(), 1000) }); document.getElementById("registerBtn").addEventListener("click",async () => { const email = document.getElementById("loginEmail").value.trim(); const password = document.getElementById("loginPassword").value.trim(); if(!email || !password){ showToast("اكتب البريد وكلمة المرور للإنشاء"); return; } const { data, error } = await supabase.auth.signUp({ email, password }) if (error) { showToast('خطأ في التسجيل: ' + error.message) return } if (data.user) { await supabase.from('profiles').insert([{ id: data.user.id, full_name: email.split('@')[0] }]) showToast('تم إنشاء الحساب وحفظ البيانات بنجاح!') setTimeout(() => location.reload(), 1000) } }); document.addEventListener("keydown",event => { if(event.key === "Escape"){ closeCart(); closeLogin(); searchBox.classList.remove("show"); } }); // تشغيل جلب البيانات أول ما تفتح الصفحة fetchProducts();
